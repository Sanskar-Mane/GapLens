# GapLens — AI-Assisted Information Gap Detection Platform
**Shinchan Online Hackathon 2026** | **Problem Statement: Digital Dark Spot**

GapLens is an analytical intelligence platform designed to uncover "Digital Dark Spots" in multi-source policies, academic regulations, government schemes, and business compliance documents. Instead of summarizing text, GapLens analyzes what is **missing, contradictory, or outdated**.

---

## 🏛️ Architecture & Highlights (Phase 0 & 1)

- **Two-Folder Architecture:** `GapLens/backend` (Express, Mongoose, Jina AI & Cheerio) and `GapLens/frontend` (React JSX, Vite, Tailwind CSS, Lucide icons).
- **Zero-Crash Resilience:** Automatic graceful fallback from MongoDB to in-memory session persistence when database is offline.
- **Dual OpenRouter Key Integration:** Supports backend `.env` (`OPENROUTER_API_KEY`) and browser-level `localStorage` overrides with model selection (`google/gemini-2.5-flash`, `meta-llama/llama-3.3-70b-instruct:free`, `deepseek/deepseek-chat-v3-0324:free`).
- **1-Click Judge Datasets:** Pre-seeded realistic scenarios with intentional dark spots:
  1. *Post-Matric Scholarship Verification:* Conflicting income limits (Rs. 2.5L vs Rs. 8L), missing renewal conditions, outdated 2023 offline submission circular.
  2. *Cloud Kitchen Licensing & Compliance:* Conflicting turnover thresholds (Rs. 10L vs Rs. 12L), missing Fire & Pollution NOCs, undated blog.
- **Multi-Source Workspace:** Dynamic collection of 2 to 6 sources with auto URL extraction via Jina AI reader and direct HTML fallback.
- **5-Panel Analytical Blueprint:** Visual layout for Corroborated Facts, Information Gaps (Dark Spots), Contradictions Radar, Temporal Warnings, and Prioritized Verification Checklist.

---

## 🚀 Quick Start Guide

### 1. Start Backend (Port 5000)
```bash
cd GapLens/backend
npm install
npm start
# Server boots on http://localhost:5000
```

### 2. Start Frontend (Port 5173)
```bash
cd GapLens/frontend
npm install
npm run dev
# Vite runs on http://localhost:5173
```

## 📡 Backend Endpoints
- `GET /` — API directory and storage mode probe
- `GET /api/health` — System health, DB status, and OpenRouter readiness
- `GET /api/scenarios` — Pre-seeded sample scenarios
- `POST /api/extract-url` — Automated web scraping via Jina AI with fallback
- `POST /api/sessions` — Persists research sessions (MongoDB or In-Memory)
- `GET /api/sessions` — Retrieves session history
- `POST /api/analyze` — Phase 1 validation and readiness check

const mongoose = require('mongoose');

let isDbConnected = false;
const inMemorySessions = [];

const connectDB = async () => {
  const uri = process.env.MONGO_URI || 'mongodb://127.0.0.1:27017/gaplens';
  
  try {
    // Attempt Mongoose connection with short timeout so server boots instantly if Mongo is absent
    const conn = await mongoose.connect(uri, {
      serverSelectionTimeoutMS: 2500,
    });
    isDbConnected = true;
    console.log(`[GapLens DB] MongoDB Connected: ${conn.connection.host}/${conn.connection.name}`);
  } catch (error) {
    isDbConnected = false;
    console.warn(`[GapLens DB Warning] MongoDB offline (${error.message}) â€” running in In-Memory Fallback Mode`);
  }
};

const getDbStatus = () => ({
  connected: isDbConnected,
  mode: isDbConnected ? 'MongoDB' : 'In-Memory Fallback',
  inMemoryCount: inMemorySessions.length
});

module.exports = {
  connectDB,
  getDbStatus,
  isDbConnected: () => isDbConnected,
  inMemorySessions
};

const mongoose = require('mongoose');

const SourceItemSchema = new mongoose.Schema({
  id: {
    type: String,
    required: true,
  },
  title: {
    type: String,
    default: '',
  },
  sourceType: {
    type: String,
    enum: [
      'Official Notice',
      'Third-Party Guide',
      'Archived Circular',
      'News Article',
      'Forum/Other'
    ],
    default: 'Official Notice',
  },
  url: {
    type: String,
    default: '',
  },
  publishedDate: {
    type: String,
    default: '',
  },
  content: {
    type: String,
    required: true,
  },
}, { _id: false });

const ResearchSessionSchema = new mongoose.Schema({
  topic: {
    type: String,
    required: [true, 'Research topic or decision goal is required'],
    trim: true,
  },
  category: {
    type: String,
    enum: ['Education', 'Careers', 'Public Services', 'Research', 'Business'],
    default: 'Education',
  },
  sources: {
    type: [SourceItemSchema],
    validate: [
      (val) => Array.isArray(val) && val.length >= 2,
      'At least two sources are required for cross-verification analysis'
    ],
  },
  modelUsed: {
    type: String,
    default: 'google/gemini-2.5-flash',
  },
  analysisResult: {
    type: mongoose.Schema.Types.Mixed,
    default: null,
  },
  createdAt: {
    type: Date,
    default: Date.now,
  },
});

module.exports = mongoose.model('ResearchSession', ResearchSessionSchema);

const express = require('express');
const axios = require('axios');
const cheerio = require('cheerio');
const sampleScenarios = require('../data/sampleScenarios');
const ResearchSession = require('../models/ResearchSession');
const { isDbConnected, inMemorySessions } = require('../config/db');

const router = express.Router();

/**
 * GET /api/health
 * Returns service status, database connectivity, and OpenRouter configuration
 */
router.get('/health', (req, res) => {
  const dbStatus = isDbConnected();
  const hasServerKey = Boolean(
    process.env.OPENROUTER_API_KEY &&
      process.env.OPENROUTER_API_KEY.trim() !== '' &&
      process.env.OPENROUTER_API_KEY !== 'your_openrouter_key_here'
  );
  const defaultModel = process.env.OPENROUTER_MODEL || 'google/gemini-2.5-flash';

  res.json({
    status: 'ok',
    timestamp: new Date().toISOString(),
    dbConnected: dbStatus,
    storageMode: dbStatus ? 'MongoDB Database' : 'In-Memory Fallback',
    hasServerOpenRouterKey: hasServerKey,
    defaultModel,
    appVersion: '1.0.0-phase1',
  });
});

/**
 * GET /api/scenarios
 * Returns pre-seeded realistic hackathon scenarios
 */
router.get('/scenarios', (req, res) => {
  res.json({
    success: true,
    count: sampleScenarios.length,
    scenarios: sampleScenarios,
  });
});

/**
 * POST /api/extract-url
 * Accepts { url } and fetches readable text content
 * Uses Jina AI reader with Cheerio fallback for high-fidelity extraction
 */
router.post('/extract-url', async (req, res) => {
  const { url } = req.body;

  if (!url || typeof url !== 'string' || !url.trim()) {
    return res.status(400).json({
      success: false,
      message: 'A valid URL is required for content extraction.',
    });
  }

  const cleanUrl = url.trim();

  // Basic validation of URL structure
  try {
    new URL(cleanUrl);
  } catch (err) {
    return res.status(400).json({
      success: false,
      message: 'Invalid URL format. Please include http:// or https://.',
    });
  }

  // Attempt 1: Jina AI Reader API (produces clean markdown & preserves tables)
  try {
    const jinaUrl = `https://r.jina.ai/${encodeURIComponent(cleanUrl)}`;
    const jinaResponse = await axios.get(jinaUrl, {
      timeout: 9000,
      headers: {
        'User-Agent': 'GapLens-Extractor/1.0 (Hackathon Digital Dark Spot)',
        Accept: 'text/plain, text/markdown',
      },
    });

    if (jinaResponse.data && typeof jinaResponse.data === 'string' && jinaResponse.data.length > 50) {
      const rawText = jinaResponse.data;
      // Derive a quick title from the first header or the URL
      const titleMatch = rawText.match(/^#\s+(.+)$/m) || rawText.match(/Title:\s*(.+)/i);
      const title = titleMatch ? titleMatch[1].trim() : cleanUrl;

      return res.json({
        success: true,
        extractedVia: 'Jina AI Reader',
        title: title.slice(0, 120),
        content: rawText.slice(0, 15000), // Cap reasonable payload for sources
        extractedAt: new Date().toISOString(),
      });
    }
  } catch (jinaErr) {
    console.warn(`[Extract Warning] Jina reader fallback triggered for ${cleanUrl}: ${jinaErr.message}`);
  }

  // Attempt 2: Direct HTTP fetch + Cheerio HTML parsing
  try {
    const directResponse = await axios.get(cleanUrl, {
      timeout: 8000,
      headers: {
        'User-Agent':
          'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
      },
    });

    const $ = cheerio.load(directResponse.data);
    // Remove unwanted elements
    $('script, style, nav, footer, noscript, svg, header').remove();

    const title = $('title').text().trim() || $('h1').first().text().trim() || cleanUrl;

    // Collect paragraph text
    const paragraphs = [];
    $('article p, main p, body p').each((_, el) => {
      const text = $(el).text().trim();
      if (text.length > 25) {
        paragraphs.push(text);
      }
    });

    const combinedText = paragraphs.join('\n\n');

    if (combinedText.length >= 40) {
      return res.json({
        success: true,
        extractedVia: 'Direct HTML Parser',
        title: title.slice(0, 120),
        content: combinedText.slice(0, 15000),
        extractedAt: new Date().toISOString(),
      });
    }
  } catch (directErr) {
    console.warn(`[Extract Warning] Direct fetch failed for ${cleanUrl}: ${directErr.message}`);
  }

  // Graceful fallback response when automated scraping fails or URL is synthetic
  return res.status(200).json({
    success: false,
    message:
      'Could not automatically scrape content from this URL (target site may require login, block scraping, or be a sample URL). You can copy & paste the raw text directly into the Source Content box below.',
    extractedAt: new Date().toISOString(),
  });
});

/**
 * POST /api/sessions
 * Saves research session with graceful fallback to in-memory store
 */
router.post('/sessions', async (req, res) => {
  try {
    const { topic, category, sources, modelUsed, analysisResult } = req.body;

    if (!topic || typeof topic !== 'string' || !topic.trim()) {
      return res.status(400).json({
        success: false,
        message: 'Topic or research goal is required.',
      });
    }

    if (!Array.isArray(sources) || sources.length < 2) {
      return res.status(400).json({
        success: false,
        message: 'At least two sources are required for cross-verification.',
      });
    }

    const sessionData = {
      topic: topic.trim(),
      category: category || 'Education',
      sources: sources.map((src, idx) => ({
        id: src.id || `src-${idx + 1}`,
        title: (src.title || '').trim(),
        sourceType: src.sourceType || 'Official Notice',
        url: (src.url || '').trim(),
        publishedDate: (src.publishedDate || '').trim(),
        content: (src.content || '').trim(),
      })),
      modelUsed: modelUsed || process.env.OPENROUTER_MODEL || 'google/gemini-2.5-flash',
      analysisResult: analysisResult || null,
      createdAt: new Date(),
    };

    if (isDbConnected()) {
      const savedDoc = await ResearchSession.create(sessionData);
      return res.status(201).json({
        success: true,
        storage: 'mongodb',
        session: savedDoc,
      });
    } else {
      const mockDoc = {
        _id: `mem-${Date.now()}-${Math.random().toString(36).substr(2, 6)}`,
        ...sessionData,
      };
      inMemorySessions.unshift(mockDoc);
      // Keep in-memory store bounded
      if (inMemorySessions.length > 50) {
        inMemorySessions.pop();
      }
      return res.status(201).json({
        success: true,
        storage: 'in-memory',
        session: mockDoc,
      });
    }
  } catch (err) {
    console.error('[Session Error]', err);
    return res.status(500).json({
      success: false,
      message: 'Failed to save research session: ' + err.message,
    });
  }
});

/**
 * GET /api/sessions
 * Retrieves saved sessions list
 */
router.get('/sessions', async (req, res) => {
  try {
    if (isDbConnected()) {
      const sessions = await ResearchSession.find()
        .sort({ createdAt: -1 })
        .limit(20)
        .lean();
      return res.json({ success: true, count: sessions.length, sessions });
    } else {
      return res.json({
        success: true,
        storage: 'in-memory',
        count: inMemorySessions.length,
        sessions: inMemorySessions,
      });
    }
  } catch (err) {
    return res.status(500).json({
      success: false,
      message: 'Failed to retrieve sessions: ' + err.message,
    });
  }
});

/**
 * POST /api/analyze
 * Phase 1 validation stub: Validates inputs, checks OpenRouter key availability,
 * and sets up workspace readiness for Phase 3 deep cross-analysis.
 */
router.post('/analyze', async (req, res) => {
  try {
    const { topic, category, sources, openRouterKey, model } = req.body;

    // 1. Validate Topic
    if (!topic || typeof topic !== 'string' || !topic.trim()) {
      return res.status(400).json({
        success: false,
        message: 'Please provide a clear research question or decision goal.',
      });
    }

    // 2. Validate Sources (Minimum 2 with content)
    if (!Array.isArray(sources) || sources.length < 2) {
      return res.status(400).json({
        success: false,
        message: 'Minimum 2 sources are required for cross-source gap detection.',
      });
    }

    const validSources = sources.filter(
      (s) => s && s.content && typeof s.content === 'string' && s.content.trim().length > 10
    );

    if (validSources.length < 2) {
      return res.status(400).json({
        success: false,
        message: `At least 2 sources must contain meaningful text (found ${validSources.length} valid source${
          validSources.length === 1 ? '' : 's'
        }). Please fill in or paste content for your sources.`,
      });
    }

    // 3. Check OpenRouter credentials (header, client body, or server .env)
    const headerKey = req.headers['x-openrouter-key'];
    const activeKey =
      (typeof openRouterKey === 'string' && openRouterKey.trim() !== '' && openRouterKey) ||
      (typeof headerKey === 'string' && headerKey.trim() !== '' && headerKey) ||
      (process.env.OPENROUTER_API_KEY &&
        process.env.OPENROUTER_API_KEY !== 'your_openrouter_key_here' &&
        process.env.OPENROUTER_API_KEY);

    const activeModel =
      model ||
      req.headers['x-openrouter-model'] ||
      process.env.OPENROUTER_MODEL ||
      'google/gemini-2.5-flash';

    // 4. Calculate source statistics and warnings
    const undatedSources = validSources.filter((s) => !s.publishedDate || !s.publishedDate.trim());
    const totalChars = validSources.reduce((acc, s) => acc + s.content.length, 0);

    return res.json({
      success: true,
      readyForPhase3: true,
      message: 'Phase 1 Source Workspace validated and prepared for AI Gap Detection.',
      validation: {
        topic: topic.trim(),
        category: category || 'Education',
        totalSources: validSources.length,
        undatedCount: undatedSources.length,
        hasUndatedWarning: undatedSources.length > 0,
        totalCharacters: totalChars,
      },
      aiConfig: {
        hasKey: Boolean(activeKey),
        keySource: openRouterKey ? 'client' : headerKey ? 'header' : activeKey ? 'server' : 'none',
        activeModel,
        mode: activeKey ? 'Live OpenRouter AI' : 'Offline Seeded Demo Mode',
      },
    });
  } catch (err) {
    console.error('[Analyze Stub Error]', err);
    return res.status(500).json({
      success: false,
      message: 'Analysis validation failed: ' + err.message,
    });
  }
});

module.exports = router;

import React, { useState, useEffect } from 'react';
import Navbar from './components/Navbar';
import ApiKeyModal from './components/ApiKeyModal';
import SampleLoader from './components/SampleLoader';
import TopicInput from './components/TopicInput';
import SourceCollector from './components/SourceCollector';
import ActionFooter from './components/ActionFooter';
import DashboardPreview from './components/DashboardPreview';
import {
  checkHealth,
  fetchScenarios,
  saveResearchSession,
  triggerAnalysisPrep
} from './services/api';
import { FALLBACK_SCENARIOS } from './data/fallbackScenarios';

const DEFAULT_SOURCES = [
  {
    id: 'src-1',
    title: 'Official Notification / Source A',
    sourceType: 'Official Notice',
    url: '',
    publishedDate: '',
    content: '',
  },
  {

    title: 'Secondary Source / Guide B',
    sourceType: 'Third-Party Guide',

    content: '',
  },

export default function App() {
  // System & API Configuration State
  const [backendHealth, setBackendHealth] = useState(null);
  const [scenarios, setScenarios] = useState(FALLBACK_SCENARIOS);
  const [openRouterKey, setOpenRouterKey] = useState(
    () => localStorage.getItem('gaplens_openrouter_key') || ''
  );
  const [activeModel, setActiveModel] = useState(
    () => localStorage.getItem('gaplens_openrouter_model') || 'google/gemini-2.5-flash'
  );
  const [forceOfflineDemo, setForceOfflineDemo] = useState(
    () => localStorage.getItem('gaplens_force_offline_demo') === 'true'
  );
  const [isKeyModalOpen, setIsKeyModalOpen] = useState(false);

  // Phase 1 Workspace Input State
  const [topic, setTopic] = useState(
    'What should I verify before applying for the State Post-Matric Scholarship?'
  );
  const [category, setCategory] = useState('Education');
  const [sources, setSources] = useState(FALLBACK_SCENARIOS[0].sources);
  const [activeScenarioId, setActiveScenarioId] = useState(FALLBACK_SCENARIOS[0].id);

  // Analysis & Validation Trigger State
  const [isAnalyzing, setIsAnalyzing] = useState(false);
  const [analysisNotice, setAnalysisNotice] = useState(null);
  const [lastAnalysisResult, setLastAnalysisResult] = useState(null);

  // 1. Check Backend Health on Mount & Periodic Interval
  useEffect(() => {
    let isMounted = true;
    const fetchStatus = async () => {
      const health = await checkHealth();
      if (isMounted) setBackendHealth(health);
    };

    const loadScenarios = async () => {
      const data = await fetchScenarios();
      if (isMounted && data && data.length > 0) {
        setScenarios(data);
      }
    };

    fetchStatus();
    loadScenarios();

    const interval = setInterval(fetchStatus, 15000);
    return () => {
      isMounted = false;
      clearInterval(interval);
    };
  }, []);

  // 2. OpenRouter Config Persistence
  const handleSaveConfig = (key, model) => {
    setOpenRouterKey(key);
    setActiveModel(model);
    localStorage.setItem('gaplens_openrouter_key', key);
    localStorage.setItem('gaplens_openrouter_model', model);
  };

  const handleToggleForceDemo = (checked) => {
    setForceOfflineDemo(checked);
    localStorage.setItem('gaplens_force_offline_demo', checked ? 'true' : 'false');
  };

  // 3. 1-Click Preset Scenario Loader
  const handleSelectScenario = (scenario) => {
    setActiveScenarioId(scenario.id);
    setTopic(scenario.topic);
    setCategory(scenario.category || 'Education');
    setSources(
      scenario.sources.map((s, idx) => ({
        ...s,
        id: s.id || `src-${idx + 1}`,
      }))
    );
    setAnalysisNotice({
      type: 'success',
      title: `Loaded Preset: ${scenario.name}`,
      message: `Scenario populated with ${scenario.sources.length} sources containing intentional flaws (${scenario.flawsSummary || 'Contradictions & Gaps'}).`,
    });
    setTimeout(() => setAnalysisNotice(null), 5000);
  };

  // 4. Source Management Handlers (2 to 6 sources)
  const handleAddSource = () => {
    if (sources.length >= 6) return;
    const newId = `src-${Date.now().toString(36)}`;
    const newSource = {
      id: newId,
      title: `Source ${sources.length + 1}`,
      sourceType: 'Third-Party Guide',
      url: '',
      publishedDate: '',
      content: '',
    };
    setSources([...sources, newSource]);
  };

  const handleUpdateSource = (sourceId, fields) => {
    setSources((prev) =>
      prev.map((s) => (s.id === sourceId ? { ...s, ...fields } : s))
    );
  };

  const handleDeleteSource = (sourceId) => {
    if (sources.length <= 2) return;
    setSources((prev) => prev.filter((s) => s.id !== sourceId));
  };

  // 5. Reset Workspace Handler
  const handleResetWorkspace = () => {
    if (window.confirm('Reset the workspace and clear all current sources?')) {
      setTopic('');
      setCategory('Education');
      setSources([
        {
          id: 'src-1',
          title: '',
          sourceType: 'Official Notice',
          url: '',
          publishedDate: '',
          content: '',
        },
        {
          id: 'src-2',
          title: '',
          sourceType: 'Third-Party Guide',
          url: '',
          publishedDate: '',
          content: '',
        },
      ]);
      setActiveScenarioId(null);
      setAnalysisNotice(null);
      setLastAnalysisResult(null);
    }
  };

  // 6. Primary Action: Validate & Trigger Phase 1 Preparation
  const handleTriggerAnalysis = async () => {
    setIsAnalyzing(true);
    setAnalysisNotice(null);

    const payload = {
      topic,
      category,
      sources,
      openRouterKey: forceOfflineDemo ? '' : openRouterKey,
      model: activeModel,
    };

    try {
      // Step A: Call validation stub
      const valResult = await triggerAnalysisPrep(payload);

      if (valResult.success) {
        // Step B: Save session to database or in-memory
        const saveResult = await saveResearchSession({
          topic,
          category,
          sources,
          modelUsed: activeModel,
        });

        setLastAnalysisResult(valResult);
        setAnalysisNotice({
          type: 'success',
          title: 'Phase 1 Inputs Validated & Synchronized',
          message: `${valResult.validation?.totalSources} sources verified (${valResult.validation?.totalCharacters} chars). Saved via ${saveResult.storage || 'Storage Engine'} on session ID ${saveResult.session?._id || 'local'}. Ready for Phase 3 deep AI cross-examination.`,
        });
      } else {
        setAnalysisNotice({
          type: 'error',
          title: 'Validation Blocked',
          message: valResult.message || 'Please check your inputs and try again.',
        });
      }
    } catch (err) {
      setAnalysisNotice({
        type: 'error',
        title: 'Execution Error',
        message: err.message || 'Failed to complete Phase 1 validation.',
      });
    } finally {
      setIsAnalyzing(false);
    }
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans selection:bg-indigo-500 selection:text-white bg-grid-pattern">
      
      {/* Top Navigation Bar */}
      <Navbar
        backendHealth={backendHealth}
        hasOpenRouterKey={Boolean(openRouterKey || backendHealth?.data?.hasServerOpenRouterKey)}
        activeModel={activeModel}
        onOpenKeyModal={() => setIsKeyModalOpen(true)}
        onResetWorkspace={handleResetWorkspace}
        forceOfflineDemo={forceOfflineDemo}
      />

      {/* Main Container */}
      <main className="flex-1 mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8 space-y-6">
        
        {/* Quick-Load Demo Scenarios for Hackathon Judges */}
        <SampleLoader
          scenarios={scenarios}
          onSelectScenario={handleSelectScenario}
          activeScenarioId={activeScenarioId}
        />

        {/* Section 1: Research Question & Decision Goal Input */}
        <TopicInput
          topic={topic}
          onTopicChange={setTopic}
          category={category}
          onCategoryChange={setCategory}
        />

        {/* Section 2: Multi-Source Collector & Card Workspace */}
        <SourceCollector
          sources={sources}
          onAddSource={handleAddSource}
          onUpdateSource={handleUpdateSource}
          onDeleteSource={handleDeleteSource}
        />

        {/* Section 3: Live Validation & Action Footer */}
        <ActionFooter
          topic={topic}
          sources={sources}
          activeModel={activeModel}
          hasOpenRouterKey={Boolean(openRouterKey || backendHealth?.data?.hasServerOpenRouterKey)}
          forceOfflineDemo={forceOfflineDemo}
          isAnalyzing={isAnalyzing}
          onTriggerAnalysis={handleTriggerAnalysis}
          analysisNotice={analysisNotice}
        />

        {/* Section 4: Visual Blueprint of the 5 Expected Output Panels */}
        <DashboardPreview lastAnalysisResult={lastAnalysisResult} />

      </main>

      {/* Footer info */}
      <footer className="border-t border-slate-900 bg-slate-950 py-6 text-center text-xs text-slate-500">
        <div className="mx-auto max-w-7xl px-4 flex flex-col sm:flex-row items-center justify-between gap-2">
          <div>
            <strong className="text-slate-400">GapLens</strong> &bull; AI-Assisted Information Gap Detection Platform
          </div>
          <div className="text-slate-400">
            Shinchan Online Hackathon 2026 &bull; Problem Statement: <span className="text-indigo-400 font-medium">Digital Dark Spot</span>
          </div>
        </div>
      </footer>

      {/* OpenRouter API Key & Model Configuration Modal */}
      <ApiKeyModal
        isOpen={isKeyModalOpen}
        onClose={() => setIsKeyModalOpen(false)}
        currentKey={openRouterKey}
        currentModel={activeModel}
        onSaveConfig={handleSaveConfig}
        backendHasKey={Boolean(backendHealth?.data?.hasServerOpenRouterKey)}
        forceOfflineDemo={forceOfflineDemo}
        onToggleForceDemo={handleToggleForceDemo}
      />

    </div>
  );
}

        type: 'success',
        text: `Successfully scraped ${result.content.length} characters via ${result.extractedVia || 'Extraction Engine'}!`,

        type: 'warning',

import axios from 'axios';
import { FALLBACK_SCENARIOS } from '../data/fallbackScenarios';

// Determine backend base URL: uses current host proxy or direct localhost:5000
const API_BASE_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000/api';

const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 12000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Interceptor to inject client-side OpenRouter credentials if stored in localStorage
apiClient.interceptors.request.use((config) => {
  const clientKey = localStorage.getItem('gaplens_openrouter_key');
  const clientModel = localStorage.getItem('gaplens_openrouter_model') || 'google/gemini-2.5-flash';

  if (clientKey && clientKey.trim()) {
    config.headers['x-openrouter-key'] = clientKey.trim();
  }
  config.headers['x-openrouter-model'] = clientModel;

  return config;
});

/**
 * Fetch backend system health and configuration status
 */
export const checkHealth = async () => {
  try {
    const res = await apiClient.get('/health');
    return {
      online: true,
      data: res.data,
    };
  } catch (err) {
    return {
      online: false,
      error: err.message,
      data: {
        status: 'offline-demo',
        dbConnected: false,
        storageMode: 'Browser In-Memory Fallback',
        hasServerOpenRouterKey: false,
        defaultModel: 'google/gemini-2.5-flash',
      },
    };
  }
};

/**
 * Fetch hackathon pre-seeded sample scenarios
 */
export const fetchScenarios = async () => {
  try {
    const res = await apiClient.get('/scenarios');
    if (res.data && res.data.scenarios && res.data.scenarios.length > 0) {
      return res.data.scenarios;
    }
    return FALLBACK_SCENARIOS;
  } catch (err) {
    console.warn('[GapLens API] Backend scenarios offline, using local fallback scenarios.');
    return FALLBACK_SCENARIOS;
  }
};

/**
 * Save current research session (persisted to MongoDB or In-Memory backend store)
 */
export const saveResearchSession = async (sessionData) => {
  try {
    const res = await apiClient.post('/sessions', sessionData);
    return res.data;
  } catch (err) {
    console.warn('[GapLens API] Session save error, caching locally in localStorage');
    const localId = `local-${Date.now()}`;
    const localRecord = { ...sessionData, _id: localId, savedLocally: true };
    const history = JSON.parse(localStorage.getItem('gaplens_local_sessions') || '[]');
    history.unshift(localRecord);
    localStorage.setItem('gaplens_local_sessions', JSON.stringify(history.slice(0, 20)));
    return {
      success: true,
      storage: 'localStorage',
      session: localRecord,
    };
  }
};

/**
 * Trigger Phase 1 analysis preparation & validation
 */
export const triggerAnalysisPrep = async (payload) => {
  const clientKey = localStorage.getItem('gaplens_openrouter_key');
  const clientModel = localStorage.getItem('gaplens_openrouter_model') || 'google/gemini-2.5-flash';

  const enrichedPayload = {
    ...payload,
    openRouterKey: clientKey || payload.openRouterKey,
    model: clientModel || payload.model,
  };

  try {
    const res = await apiClient.post('/analyze', enrichedPayload);
    return res.data;
  } catch (err) {
    if (err.response?.data) {
      return err.response.data;
    }
    // Fallback validation stub on frontend
    const validSources = (payload.sources || []).filter(
      (s) => s.content && s.content.trim().length > 10
    );
    const undated = validSources.filter((s) => !s.publishedDate || !s.publishedDate.trim());

    if (!payload.topic || !payload.topic.trim()) {
      return { success: false, message: 'Please enter a research topic or decision goal.' };
    }
    if (validSources.length < 2) {
      return { success: false, message: 'At least 2 sources with text are required.' };
    }

    return {
      success: true,
      readyForPhase3: true,
      message: 'Workspace validated via Client Fallback Engine.',
      validation: {
        topic: payload.topic,
        category: payload.category || 'Education',
        totalSources: validSources.length,
        undatedCount: undated.length,
        hasUndatedWarning: undated.length > 0,
        totalCharacters: validSources.reduce((acc, s) => acc + s.content.length, 0),
      },
      aiConfig: {
        hasKey: Boolean(clientKey),
        keySource: clientKey ? 'client' : 'none',
        activeModel: clientModel,
        mode: clientKey ? 'Live OpenRouter AI' : 'Offline Seeded Demo Mode',
      },
    };
  }
};

/**
 * Direct OpenRouter probe (for checking key validity in modal)
 */
export const probeOpenRouter = async (key, model = 'google/gemini-2.5-flash') => {
  if (!key || !key.trim()) {
    return { success: false, message: 'API key is empty.' };
  }

  try {
    const res = await axios.post(
      'https://openrouter.ai/api/v1/chat/completions',
      {
        model,
        messages: [{ role: 'user', content: 'Say "GapLens OpenRouter Ready" in 4 words.' }],
        max_tokens: 15,
      },
      {
        headers: {
          Authorization: `Bearer ${key.trim()}`,
          'Content-Type': 'application/json',
          'HTTP-Referer': 'https://gaplens.local',
          'X-Title': 'GapLens Hackathon Platform',
        },
        timeout: 10000,
      }
    );

    const reply = res.data?.choices?.[0]?.message?.content || 'Verified successfully';
    return {
      success: true,
      reply,
      modelUsed: res.data?.model || model,
    };
  } catch (err) {
    const msg =
      err.response?.data?.error?.message ||
      err.response?.data?.message ||
      err.message ||
      'Failed to reach OpenRouter API';
    return { success: false, message: msg };
  }
};
